len(vgroups) # Filter by emptiness. for idx, vg in enumerate(vgroups): if vgroups_empty[vg.index][0]: flt_flags[idx] |= self.VGROUP_EMPTY if self.use_filter_empty and self.use_filter_empty_reverse: flt_flags[idx] &= ~self.bitflag_filter_item elif self.use_filter_empty and not self.use_filter_empty_reverse: flt_flags[idx] &= ~self.bitflag_filter_item # Reorder by name or average weight. if
